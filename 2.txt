def _format_duration(total_seconds, show_seconds=True):
    """تابع کمکی برای فرمت کردن زمان به صورت ساعت، دقیقه و ثانیه."""
    if total_seconds < 0: total_seconds = 0
    
    hours, remainder = divmod(total_seconds, 3600)
    minutes, seconds = divmod(remainder, 60)
    
    parts = []
    if int(hours) > 0:
        parts.append(f"{int(hours)} ساعت")
    if int(minutes) > 0:
        parts.append(f"{int(minutes)} دقیقه")
    if show_seconds and int(seconds) > 0:
        parts.append(f"{int(seconds)} ثانیه")

    if not parts:
        return "0 دقیقه" if not show_seconds else "0 ثانیه"

    return " و ".join(parts)

def generate_html_report(entries):
    if not entries:
        return HTML_TEMPLATE.format(
            daily_summary_section="<li>هیچ فعالیتی یافت نشد.</li>",
            summary_section="<li>هیچ فعالیتی یافت نشد.</li>",
            details_section="<p>داده‌ای یافت نشد.</p>",
            chart_labels="[]",
            chart_data="[]"
        )

    summary_tasks = {}
    summary_categories = {}
    details_by_date = {}

    for entry in entries:
        _, task_name, category, start_str, end_str, notes = entry
        category = category if category else 'بدون دسته'
        if not start_str or not end_str: continue

        start_dt = datetime.strptime(start_str.split('.')[0], '%Y-%m-%d %H:%M:%S')
        end_dt = datetime.strptime(end_str.split('.')[0], '%Y-%m-%d %H:%M:%S')
        duration_seconds = (end_dt - start_dt).total_seconds()
        
        summary_tasks[task_name] = summary_tasks.get(task_name, 0) + duration_seconds
        summary_categories[category] = summary_categories.get(category, 0) + duration_seconds
        
        date_key = start_dt.date()
        if date_key not in details_by_date:
            details_by_date[date_key] = []
        details_by_date[date_key].append({
            'start_time': start_dt.strftime('%H:%M'), 'end_time': end_dt.strftime('%H:%M'),
            'duration_seconds': duration_seconds, 'task_name': task_name,
            'notes': notes if notes and notes.strip() else None
        })

    summary_html = ""
    for task, total_seconds in sorted(summary_tasks.items()):
        summary_html += f"<li>مجموع '{task}': {_format_duration(total_seconds, show_seconds=False)}</li>"

    sorted_dates = sorted(details_by_date.keys())
    
    daily_summary_html = ""
    for date_obj in sorted_dates:
        jalali_date = jdatetime.date.fromgregorian(date=date_obj)
        english_day_name = date_obj.strftime('%A')
        persian_day_name = PERSIAN_WEEKDAYS.get(english_day_name, english_day_name)
        total_seconds_today = sum(item['duration_seconds'] for item in details_by_date[date_obj])
        
        daily_summary_html += (
            f"<li>🗓️ {jalali_date.strftime('%Y/%m/%d')} {persian_day_name}: "
            f"{_format_duration(total_seconds_today)}</li>"
        )
    
    details_html = ""
    for date_obj in sorted_dates:
        jalali_date = jdatetime.date.fromgregorian(date=date_obj)
        english_day_name = date_obj.strftime('%A')
        persian_day_name = PERSIAN_WEEKDAYS.get(english_day_name, english_day_name)
        total_seconds_today = sum(item['duration_seconds'] for item in details_by_date[date_obj])
        
        details_html += (
            f'<div class="date-group">'
            f'<p class="date-header">🗓️ {jalali_date.strftime("%Y/%m/%d")} {persian_day_name}</p>'
            f'<p class="daily-summary">مجموع فعالیت‌های روز: {_format_duration(total_seconds_today, show_seconds=False)}</p>'
            '<ul>'
        )
        
        sorted_entries = sorted(details_by_date[date_obj], key=lambda x: x['start_time'])
        for item in sorted_entries:
            duration_text = _format_duration(item['duration_seconds'], show_seconds=True)
            
            details_html += (
                f"<li>{item['task_name']} ({duration_text}) | "
                f"از {item['start_time']} تا {item['end_time']}</li>"
            )
            if item['notes']:
                details_html += f"<li class='notes'>یادداشت: {item['notes']}</li>"
        details_html += "</ul></div>"

    chart_labels = list(summary_categories.keys())
    chart_data = [round(v / 3600, 2) for v in summary_categories.values()]

    return HTML_TEMPLATE.format(
        daily_summary_section=daily_summary_html,
        summary_section=summary_html,
        details_section=details_html,
        chart_labels=json.dumps(chart_labels, ensure_ascii=False),
        chart_data=json.dumps(chart_data)
    )

def export_to_csv(entries, file_path):
    """
    داده‌ها را با استفاده از ماژول استاندارد csv به فایل CSV منتقل می‌کند.
    """
    headers = ['id', 'task_name', 'category', 'start_time', 'end_time', 'notes']
    
    with open(file_path, 'w', newline='', encoding='utf-8-sig') as csvfile:
        writer = csv.writer(csvfile)
        writer.writerow(headers)
        writer.writerows(entries)
        
    return f"گزارش با موفقیت در {file_path} ذخیره شد."

class CategoryManagerWindow(Toplevel):
    def __init__(self, master):
        super().__init__(master)

        self.transient(master)
        self.grab_set()

        self.geometry(f'{width}x{height}+{x}+{y}')
        
        self.listbox = ttkb.Treeview(self, columns=("id", "name"), show="headings", height=10)
        self.listbox.heading("id", text="شناسه")
        self.listbox.heading("name", text="نام دسته")
        self.listbox.column("id", width=80, anchor="center")
        self.listbox.column("name", width=200, anchor="e")
        self.listbox.pack(padx=10, pady=10)
        
        self.refresh_listbox()
        
        btn_frame = ttkb.Frame(self)
        btn_frame.pack(pady=5)
        
        ttkb.Button(btn_frame, text="حذف", command=self.delete_selected, bootstyle=styles.BUTTON_DANGER).pack(side="right", padx=5)
        ttkb.Button(btn_frame, text="ویرایش", command=self.edit_selected, bootstyle=styles.BUTTON_WARNING).pack(side="right", padx=5)
        ttkb.Button(btn_frame, text="افزودن", command=self.add_new, bootstyle=styles.BUTTON_SUCCESS).pack(side="right", padx=5)

    def refresh_listbox(self):
        for row in self.listbox.get_children():
            self.listbox.delete(row)
        for cat_id, name in db.get_categories():
            self.listbox.insert("", "end", values=(cat_id, name))

    def add_new(self):
        name = simpledialog.askstring("افزودن دسته", "نام دسته جدید را وارد کنید:", parent=self)
        if name and name.strip():
            if db.get_category_id_by_name(name.strip()) is None:
                db.add_category(name.strip())
                self.refresh_listbox()
                self.master.populate_categories()
                messagebox.showinfo("موفقیت", "دسته جدید با موفقیت افزوده شد.", parent=self)
            else:
                messagebox.showerror("خطا", f"نام '{name.strip()}' از قبل وجود دارد.", parent=self)

    def edit_selected(self):
        selected_item = self.listbox.selection()
        if not selected_item:
            messagebox.showwarning("انتخاب نشده", "لطفاً ابتدا یک دسته را برای ویرایش انتخاب کنید.", parent=self)
            return
            
        item_values = self.listbox.item(selected_item[0])['values']
        cat_id = item_values[0]
        old_name = item_values[1]
        
        new_name = simpledialog.askstring("ویرایش دسته", f"نام جدید را برای '{old_name}' وارد کنید:",
                                          initialvalue=old_name, parent=self)
        
        if new_name and new_name.strip() and new_name.strip() != old_name:
            success = db.update_category_name(cat_id, new_name.strip())
            if success:
                self.refresh_listbox()
                self.master.populate_categories()
                messagebox.showinfo("موفقیت", "دسته با موفقیت ویرایش شد.", parent=self)
            else:
                messagebox.showerror("خطا", f"نام '{new_name.strip()}' از قبل وجود دارد و نمی‌تواند تکراری باشد.", parent=self)

    def delete_selected(self):
        selected_item = self.listbox.selection()
        if not selected_item:
            messagebox.showwarning("انتخاب نشده", "لطفاً ابتدا یک دسته را برای حذف انتخاب کنید.", parent=self)
            return
            
        cat_id = self.listbox.item(selected_item[0])['values'][0]
        cat_name = self.listbox.item(selected_item[0])['values'][1]
        
        if db.is_category_used(cat_id):
            messagebox.showerror("خطا", f"دسته '{cat_name}' در فعالیت‌ها استفاده شده و قابل حذف نیست.", parent=self)
            return
            
        if messagebox.askyesno("تایید حذف", f"آیا مطمئن هستید می‌خواهید دسته '{cat_name}' را حذف کنید؟", parent=self):
            db.delete_category(cat_id)
            self.refresh_listbox()
            self.master.populate_categories()
            messagebox.showinfo("موفقیت", "دسته با موفقیت حذف شد.", parent=self)

        ttkb.Button(btn_frame, text="خروجی CSV", command=self.export_csv, bootstyle=styles.BUTTON_SECONDARY).pack(side="right", padx=5)

    def _get_gregorian_date_range_from_inputs(self):
        try:
            start_jalali_str = self.start_date_entry.get()
            end_jalali_str = self.end_date_entry.get()
            if "Y" in start_jalali_str or "Y" in end_jalali_str: return None, None

            start_gregorian = jdatetime.datetime.strptime(start_jalali_str, '%Y/%m/%d').togregorian().date()
            end_gregorian = jdatetime.datetime.strptime(end_jalali_str, '%Y/%m/%d').togregorian().date()
            
            start_date_for_db = start_gregorian.strftime('%Y-%m-%d')
            end_date_for_db = (end_gregorian + timedelta(days=1)).strftime('%Y-%m-%d')
            
            return start_date_for_db, end_date_for_db
        except (ValueError, AttributeError):
            messagebox.showerror("خطای فرمت", "فرمت تاریخ شمسی صحیح نیست. لطفاً از فرمت YYYY/MM/DD استفاده کنید.", parent=self)
            return None, None

    def generate_and_show_report(self):
        start_date, end_date = self._get_gregorian_date_range_from_inputs()
        if not start_date:
            return

        entries = db.get_entries_by_date_range(start_date, end_date)

        if not entries:
            messagebox.showinfo("اطلاعات", "داده‌ای برای ساخت گزارش در این بازه زمانی وجود ندارد.", parent=self)
            return

        report_html = reports.generate_html_report(entries)
        
        try:
            output_dir = "reports_output"
            os.makedirs(output_dir, exist_ok=True)
            timestamp = datetime.now().strftime("%Y-%m-%d_%H-%M-%S")
            file_name = f"report_{timestamp}.html"
            file_path = os.path.join(output_dir, file_name)

            with open(file_path, 'w', encoding='utf-8-sig') as f:
                f.write(report_html)
            
            webbrowser.open('file://' + os.path.realpath(file_path))
            messagebox.showinfo("گزارش ساخته شد", f"گزارش با موفقیت در پوشه 'reports_output' ذخیره شد.", parent=self)
        except Exception as e:
            messagebox.showerror("خطا", f"خطا در ساخت یا باز کردن فایل گزارش: {e}", parent=self)

    def export_csv(self):
        start_date, end_date = self._get_gregorian_date_range_from_inputs()
        if not start_date:
            return

        entries = db.get_entries_by_date_range(start_date, end_date)
        if not entries:
            messagebox.showinfo("اطلاعات", "داده‌ای برای خروجی در این بازه زمانی وجود ندارد.", parent=self)
            return

        file_path = filedialog.asksaveasfilename(defaultextension=".csv", 
                                                 filetypes=[("CSV files", "*.csv")])
        if file_path:
            message = reports.export_to_csv(entries, file_path)
            messagebox.showinfo("موفقیت", message, parent=self)

            end_gregorian = datetime.strptime(end_time_str, '%Y-%m-%d %H:%M:%S')
            end_jalali = jdatetime.datetime.fromgregorian(datetime=end_gregorian)
            self.end_entry.insert(0, end_jalali.strftime('%Y/%m/%d %H:%M:%S'))
        self.end_entry.grid(row=3, column=0, padx=5, pady=5, sticky='ew')
        
        ttkb.Label(frame, text="یادداشت").grid(row=4, column=1, padx=5, pady=5, sticky='e')
        self.notes_text = Text(frame, width=30, height=4, font=styles.DEFAULT_FONT)
        self.notes_text.grid(row=4, column=0, padx=5, pady=5, sticky='ew')
        if notes:
            self.notes_text.insert('1.0', notes)
        
        ttkb.Button(frame, text="ذخیره تغییرات", 
                      command=self.save_changes, 
                      bootstyle=styles.BUTTON_PRIMARY).grid(row=5, column=0, columnspan=2, pady=10)

    def save_changes(self):
        try:
            new_task = self.task_entry.get()
            new_cat_name = self.cat_combo.get()
            new_cat_id = self.master.category_map.get(new_cat_name)
            
            start_jalali_str = self.start_entry.get()
            start_gregorian = jdatetime.datetime.strptime(start_jalali_str, '%Y/%m/%d %H:%M:%S').togregorian()
            new_start = start_gregorian.strftime('%Y-%m-%d %H:%M:%S')

            end_jalali_str = self.end_entry.get()
            new_end = None
            if end_jalali_str.strip():
                end_gregorian = jdatetime.datetime.strptime(end_jalali_str, '%Y/%m/%d %H:%M:%S').togregorian()
                new_end = end_gregorian.strftime('%Y-%m-%d %H:%M:%S')
            
            new_notes = self.notes_text.get('1.0', 'end-1c')
            
            if not new_task:
                messagebox.showerror("خطا", "عنوان فعالیت نمی‌تواند خالی باشد.", parent=self)
                return
                
            db.update_entry(self.entry_id, new_task, new_cat_id, new_start, new_end, new_notes)
            self.master.refresh_main_log()
            self.destroy()
            
        except ValueError:
            messagebox.showerror("خطای فرمت", "فرمت تاریخ یا زمان شمسی صحیح نیست.\nفرمت صحیح: YYYY/MM/DD HH:MM:SS", parent=self)

    def save_new_entry(self):
        try:
            task = self.task_entry.get()
            cat_name = self.cat_combo.get()
            cat_id = self.master.category_map.get(cat_name)

            start_jalali_str = self.start_entry.get()
            start_gregorian = jdatetime.datetime.strptime(start_jalali_str, '%Y/%m/%d %H:%M:%S').togregorian()
            start_time_for_db = start_gregorian.strftime('%Y-%m-%d %H:%M:%S')

            end_jalali_str = self.end_entry.get()
            end_gregorian = jdatetime.datetime.strptime(end_jalali_str, '%Y/%m/%d %H:%M:%S').togregorian()
            end_time_for_db = end_gregorian.strftime('%Y-%m-%d %H:%M:%S')
            
            notes = self.notes_text.get('1.0', 'end-1c')
            
            if not task:
                messagebox.showerror("خطا", "عنوان فعالیت نمی‌تواند خالی باشد.", parent=self)
                return
                
            if start_gregorian >= end_gregorian:
                messagebox.showerror("خطا", "زمان پایان باید بعد از زمان شروع باشد.", parent=self)
                return
                
            db.add_manual_entry(task, cat_id, start_time_for_db, end_time_for_db, notes)
            self.master.refresh_main_log()
            self.destroy()
            
        except ValueError:
            messagebox.showerror("خطای فرمت", "فرمت تاریخ یا زمان شمسی صحیح نیست.\nفرمت صحیح: YYYY/MM/DD HH:MM:SS", parent=self)
